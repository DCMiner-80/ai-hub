+/// Download a file with resume support into `tmp_path`. Returns the final path on success.
+fn download_with_resume(client: &Client, url: &str, tmp_path: &Path, total_size: u64, progress_cb: &Box<ProgressCb>, concurrency: usize) -> Result<()> {
+    // If server supports range requests, we will download in parallel chunks.
+    // Fallback to single-threaded streaming if not.
+    // Check if server supports range by sending a HEAD request.
+    let head = client.head(url).send().context("HEAD request failed")?;
+    let accept_ranges = head.headers().get("accept-ranges").map(|v| v.to_str().unwrap_or("").to_lowercase()).unwrap_or_default();
+    let supports_range = accept_ranges.contains("bytes") || head.status().is_success();
+
+    if supports_range && total_size > 4 * 1024 * 1024 && concurrency > 1 {
+        // Parallel chunked download
+        // Create tmp file with full size
+        let mut f = OpenOptions::new().create(true).write(true).open(tmp_path)?;
+        f.set_len(total_size)?;
+        drop(f);
+
+        let chunk_size = (total_size as usize + concurrency - 1) / concurrency;
+        let client = client.clone();
+        let tmp_path = tmp_path.to_path_buf();
+        let progress = Arc::new(Mutex::new(0u64));
+
+        let mut handles = Vec::new();
+        for i in 0..concurrency {
+            let start = (i * chunk_size) as u64;
+            let end = min(((i + 1) * chunk_size) as u64 - 1, total_size - 1);
+            if start > end { continue; }
+            let client = client.clone();
+            let tmp_path = tmp_path.clone();
+            let progress = progress.clone();
+            let progress_cb = progress_cb.clone();
+            let url = url.to_string();
+            let handle = thread::spawn(move || -> Result<()> {
+                let range_header = format!("bytes={}-{}", start, end);
+                let mut resp = client.get(&url).header(RANGE, range_header).send().context("range request failed")?;
+                if !resp.status().is_success() && resp.status() != reqwest::StatusCode::PARTIAL_CONTENT {
+                    anyhow::bail!("range request returned {}", resp.status());
+                }
+                let mut out = OpenOptions::new().write(true).open(&tmp_path)?;
+                out.seek(SeekFrom::Start(start))?;
+                let mut buf = [0u8; 8192];
+                let mut local_read = 0u64;
+                loop {
+                    let n = resp.read(&mut buf)?;
+                    if n == 0 { break; }
+                    out.write_all(&buf[..n])?;
+                    local_read += n as u64;
+                    let mut g = progress.lock().unwrap();
+                    *g += n as u64;
+                    (progress_cb)(*g, total_size);
+                }
+                Ok(())
+            });
+            handles.push(handle);
+        }
+
+        for h in handles {
+            let r = h.join().map_err(|e| anyhow::anyhow!("thread join error"))?;
+            r?;
+        }
+        Ok(())
+    } else {
+        // Single-stream download with resume
+        let mut downloaded = 0u64;
+        if tmp_path.exists() {
+            downloaded = tmp_path.metadata()?.len();
+        }
+        let mut req = client.get(url);
+        if downloaded > 0 {
+            let range = format!("bytes={}-", downloaded);
+            req = req.header(RANGE, range);
+        }
+        let mut resp = req.send().context("download request failed")?;
+        if !(resp.status().is_success() || resp.status() == reqwest::StatusCode::PARTIAL_CONTENT) {
+            anyhow::bail!("download failed with status {}", resp.status());
+        }
+        let mut out = OpenOptions::new().create(true).append(true).open(tmp_path)?;
+        let mut buf = [0u8; 8192];
+        loop {
+            let n = resp.read(&mut buf)?;
+            if n == 0 { break; }
+            out.write_all(&buf[..n])?;
+            downloaded += n as u64;
+            (progress_cb)(downloaded, total_size);
+        }
+        Ok(())
+    }
+}

+/// Download and extract model with resume and parallel chunk support.
+pub fn download_and_extract_model(
+    manifest: &ModelsManifest,
+    model_id: &str,
+    dest_dir: &Path,
+    progress_cb: Box<ProgressCb>,
+) -> Result<()> {
+    let entry = manifest.models.iter().find(|m| m.id == model_id).context("model id not found in manifest")?;
+    let client = Client::builder().timeout(Duration::from_secs(600)).build()?;
+
+    fs::create_dir_all(dest_dir)?;
+    let tmp = dest_dir.join(format!("{}.download", entry.id));
+
+    // If server provides content-length, use it; otherwise fall back to manifest size
+    let total_size = entry.size_bytes;
+
+    // Try download with resume and parallel chunks (4 threads by default)
+    let concurrency = 4usize;
+    download_with_resume(&client, &entry.url, &tmp, total_size, &progress_cb, concurrency)?;
+
+    // verify checksum
+    let mut f = File::open(&tmp)?;
+    let mut hasher = Sha256::new();
+    let mut buf = [0u8; 8192];
+    loop {
+        let n = f.read(&mut buf)?;
+        if n == 0 { break; }
+        hasher.update(&buf[..n]);
+    }
+    let digest = hasher.finalize();
+    let hex = hex::encode(digest);
+    if !hex.eq_ignore_ascii_case(&entry.sha256) {
+        fs::remove_file(&tmp).ok();
+        anyhow::bail!("checksum mismatch: expected {}, got {}", entry.sha256, hex);
+    }
+
+    // extract zip
+    let zipf = File::open(&tmp)?;
+    let mut archive = ZipArchive::new(zipf)?;
+    for i in 0..archive.len() {
+        let mut file = archive.by_index(i)?;
+        let outpath = dest_dir.join(file.sanitized_name());
+        if (&*file.name()).ends_with('/') {
+            fs::create_dir_all(&outpath)?;
+        } else {
+            if let Some(p) = outpath.parent() {
+                fs::create_dir_all(p)?;
+            }
+            let mut outfile = File::create(&outpath)?;
+            io::copy(&mut file, &mut outfile)?;
+        }
+    }
+
+    fs::remove_file(&tmp)?;
+    Ok(())
+}

+/// Messages sent from background downloader to UI
+#[derive(Debug, Clone)]
+pub enum DownloadMsg {
+    Progress { id: String, downloaded: u64, total: u64 },
+    Done { id: String, ok: Result<()> },
+}
+
+/// Spawn a background download for a single model. Sends progress and done messages to tx.
+pub fn spawn_download(manifest_path: PathBuf, model_id: String, dest_dir: PathBuf, tx: Sender<DownloadMsg>) {
+    thread::spawn(move || {
+        let manifest = match ModelsManifest::from_file(&manifest_path) {
+            Ok(m) => m,
+            Err(e) => {
+                let _ = tx.send(DownloadMsg::Done { id: model_id.clone(), ok: Err(e.into()) });
+                return;
+            }
+        };
+
+        let cb = move |downloaded: u64, total: u64| {
+            let _ = tx.send(DownloadMsg::Progress { id: model_id.clone(), downloaded, total });
+        };
+
+        let res = download_and_extract_model(&manifest, &model_id, &dest_dir, Box::new(cb));
+        let _ = tx.send(DownloadMsg::Done { id: model_id.clone(), ok: res });
+    });
+}

+use std::sync::{Arc, Mutex};
+use std::thread;
+use std::time::Duration;
+use std::collections::VecDeque;
+use std::path::PathBuf;
+use crate::downloader_ui::{spawn_download, DownloadMsg};
+use std::sync::mpsc::{Sender, Receiver, channel};
+
+/// Simple scheduler that queues model downloads and runs them one at a time (configurable).
+pub struct DownloadScheduler {
+    queue: Arc<Mutex<VecDeque<String>>>,
+    manifest: PathBuf,
+    dest_dir: PathBuf,
+    tx_ui: Sender<DownloadMsg>,
+}
+
+impl DownloadScheduler {
+    pub fn new(manifest: PathBuf, dest_dir: PathBuf, tx_ui: Sender<DownloadMsg>) -> Self {
+        Self {
+            queue: Arc::new(Mutex::new(VecDeque::new())),
+            manifest,
+            dest_dir,
+            tx_ui,
+        }
+    }
+
+    /// Add a model id to the queue
+    pub fn enqueue(&self, model_id: String) {
+        let mut q = self.queue.lock().unwrap();
+        q.push_back(model_id);
+    }
+
+    /// Start background worker that processes the queue
+    pub fn start(self: Arc<Self>) {
+        let q = self.queue.clone();
+        let manifest = self.manifest.clone();
+        let dest = self.dest_dir.clone();
+        let tx = self.tx_ui.clone();
+        thread::spawn(move || {
+            loop {
+                let next = {
+                    let mut g = q.lock().unwrap();
+                    g.pop_front()
+                };
+                if let Some(model_id) = next {
+                    spawn_download(manifest.clone(), model_id, dest.clone(), tx.clone());
+                    // wait a bit before checking next to avoid tight loop; actual completion is signaled via UI messages
+                    thread::sleep(Duration::from_secs(2));
+                } else {
+                    thread::sleep(Duration::from_secs(1));
+                }
+            }
+        });
+    }
+}
